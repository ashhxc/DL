# ==============================
# CNN - FASHION MNIST
# ==============================


# 1. Import Libraries

import tensorflow as tf
import matplotlib.pyplot as plt
from tensorflow.keras import layers, models



# 2. Load Dataset

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.fashion_mnist.load_data()

class_names = ['T-shirt/top', 'Trouser', 'Pullover', 'Dress', 'Coat',
               'Sandal', 'Shirt', 'Sneaker', 'Bag', 'Ankle boot']



# 3. Display Original Image BEFORE Model Building

plt.imshow(x_train[0], cmap='gray')
plt.title("Original Image: " + class_names[y_train[0]])
plt.axis('off')
plt.show()



# 4. Normalize Images

x_train = x_train / 255.0
x_test = x_test / 255.0



# 5. Build CNN Model

model = models.Sequential([
    layers.Input(shape=(28, 28, 1)),
    layers.Conv2D(32, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),
    layers.Conv2D(64, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),
    layers.Flatten(),
    layers.Dense(64, activation='relu'),
    layers.Dense(10, activation='softmax')
])



# 6. Compile Model

model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)



# 7. Train Model

model.fit(
    x_train,
    y_train,
    epochs=5,
    validation_split=0.1
)



# 8. Evaluate Model

loss, accuracy = model.evaluate(x_test, y_test)
print("Test Accuracy:", accuracy)



# 9. Display Image AFTER Model Building and Training

prediction = model.predict(x_test[:1])
predicted_class = prediction.argmax()

plt.imshow(x_test[0], cmap='gray')
plt.title("Predicted: " + class_names[predicted_class] +
          "\nActual: " + class_names[y_test[0]])
plt.axis('off')
plt.show()
